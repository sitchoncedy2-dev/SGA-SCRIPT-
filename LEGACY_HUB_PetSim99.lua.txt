local function char() return LP.Character end
local function hrp() local c = char() return c and c:FindFirstChild("HumanoidRootPart") end
local function hum() local c = char() return c and c:FindFirstChildOfClass("Humanoid") end

local function collectRemotes()
	table.clear(remotes)
	local function add(inst)
		if inst and (inst:IsA("RemoteEvent") or inst:IsA("RemoteFunction")) then
			table.insert(remotes, inst)
		end
	end
	for _, d in ipairs(ReplicatedStorage:GetDescendants()) do
		if d:IsA("RemoteEvent") or d:IsA("RemoteFunction") then add(d) end
	end
	return #remotes
end

local function fireNames(names, args)
	args = args or {}
	if #remotes == 0 then collectRemotes() end
	for _, name in ipairs(names) do
		for _, r in ipairs(remotes) do
			pcall(function()
				if r:IsA("RemoteEvent") then
					r:FireServer(name, unpack(args))
					r:FireServer(name, args)
					if #args > 0 then r:FireServer(unpack(args)) end
				else
					r:InvokeServer(name, unpack(args))
					r:InvokeServer(name, args)
				end
			end)
		end
		pcall(function()
			local net = ReplicatedStorage:FindFirstChild("Network")
				or ReplicatedStorage:FindFirstChild("Networking")
			if net then
				local child = net:FindFirstChild(name) or net:FindFirstChild(name, true)
				if child then
					if child:IsA("RemoteEvent") then child:FireServer(unpack(args)) end
					if child:IsA("RemoteFunction") then child:InvokeServer(unpack(args)) end
				end
			end
		end)
		pcall(function()
			local lib = ReplicatedStorage:FindFirstChild("Library")
			if lib then
				local net = lib:FindFirstChild("Network") or lib:FindFirstChild("Networking")
				if net then
					local fn = net:FindFirstChild(name)
					if fn then
						if fn:IsA("RemoteEvent") then fn:FireServer(unpack(args)) end
						if fn:IsA("RemoteFunction") then fn:InvokeServer(unpack(args)) end
					end
				end
			end
		end)
	end
end

-- movement
local speedConn
local function applySpeed(on)
	S.speed = on
	if speedConn then pcall(function() speedConn:Disconnect() end) speedConn = nil end
	if on then
		speedConn = track(RunService.Heartbeat:Connect(function()
			if not S.speed then return end
			local h = hum()
			if h then h.WalkSpeed = S.speedVal end
		end))
	else
		local h = hum()
		if h then h.WalkSpeed = 16 end
	end
end

local function applyJump(on)
	S.jump = on
	local h = hum()
	if h then
		pcall(function() h.UseJumpPower = true end)
		h.JumpPower = on and S.jumpVal or 50
	end
end

local flyBV, flyBG, flyConn
local function applyFly(on)
	S.fly = on
	if flyBV then pcall(function() flyBV:Destroy() end) flyBV = nil end
	if flyBG then pcall(function() flyBG:Destroy() end) flyBG = nil end
	if flyConn then pcall(function() flyConn:Disconnect() end) flyConn = nil end
	local r = hrp()
	if not on or not r then return end
	flyBV = Instance.new("BodyVelocity")
	flyBV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
	flyBV.Velocity = Vector3.zero
	flyBV.Parent = r
	flyBG = Instance.new("BodyGyro")
	flyBG.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
	flyBG.P = 2500
	flyBG.Parent = r
	flyConn = track(RunService.RenderStepped:Connect(function()
		if not S.fly or not flyBV then return end
		local cam = Workspace.CurrentCamera
		if not cam then return end
		local d = Vector3.zero
		if UIS:IsKeyDown(Enum.KeyCode.W) then d = d + cam.CFrame.LookVector end
		if UIS:IsKeyDown(Enum.KeyCode.S) then d = d - cam.CFrame.LookVector end
		if UIS:IsKeyDown(Enum.KeyCode.A) then d = d - cam.CFrame.RightVector end
		if UIS:IsKeyDown(Enum.KeyCode.D) then d = d + cam.CFrame.RightVector end
		if UIS:IsKeyDown(Enum.KeyCode.Space) then d = d + Vector3.new(0, 1, 0) end
		if UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.LeftShift) then
			d = d - Vector3.new(0, 1, 0)
		end
		if d.Magnitude > 0 then d = d.Unit * S.flySpd end
		flyBV.Velocity = d
		if flyBG then flyBG.CFrame = cam.CFrame end
	end))
end

local function nearestBreakable()
	local my = hrp()
	if not my then return nil end
	local best, bestDist = nil, 1e9
	local roots = {Workspace:FindFirstChild("__THINGS"), Workspace:FindFirstChild("Breakables"), Workspace}
	for _, root in ipairs(roots) do
		if root then
			for _, obj in ipairs(root:GetDescendants()) do
				if obj:IsA("BasePart") then
					local n = obj.Name:lower()
					if n:find("break") or n:find("coin") or n:find("chest") or n:find("candy")
						or n:find("pumpkin") or n:find("orb") or n:find("loot") or n:find("block") then
						local d = (obj.Position - my.Position).Magnitude
						if d < bestDist and d < 150 then
							bestDist = d
							best = obj
						end
					end
				end
			end
		end
	end
	return best
end

local function farmTick()
	local my = hrp()
	if not my then return end
	local target = nearestBreakable()
	if target then
		pcall(function() my.CFrame = CFrame.new(target.Position + Vector3.new(0, 4, 0)) end)
	end
	fireNames({"Tap", "Click", "Break", "DamageBreakable", "HitBreakable", "Farm", "DealDamage", "PetTap"}, {})
	local c = char()
	if c then
		local tool = c:FindFirstChildOfClass("Tool")
		if tool then pcall(function() tool:Activate() end) end
	end
	pcall(function()
		local cam = Workspace.CurrentCamera
		if cam then
			local vp = cam.ViewportSize
			VirtualInputManager:SendMouseButtonEvent(vp.X / 2, vp.Y / 2, 0, true, game, 0)
			VirtualInputManager:SendMouseButtonEvent(vp.X / 2, vp.Y / 2, 0, false, game, 0)
		end
	end)
end

local function collectTick()
	local my = hrp()
	if not my then return end
	fireNames({
		"CollectOrb", "CollectOrbs", "ClaimOrb", "PickupOrb", "Orbs_Collect",
		"CollectAllOrbs", "MagnetOrbs", "CollectLootbag", "CollectLoot",
		"Pickup", "Collect", "Magnet", "CollectAll", "CollectCandy",
		"CollectLuckOrb", "LuckOrb", "ClaimLuck",
	}, {})
	local things = Workspace:FindFirstChild("__THINGS") or Workspace
	local range = S.collectRange or 120
	local bestOrb, bestDist = nil, 1e9
	for _, obj in ipairs(things:GetDescendants()) do
		if obj:IsA("BasePart") or obj:IsA("MeshPart") then
			local n = obj.Name:lower()
			if n:find("orb") or n:find("luck") or n:find("loot") or n:find("bag")
				or n:find("candy") or n:find("corn") or n:find("gem") or n:find("burst") then
				local d = (obj.Position - my.Position).Magnitude
				if d < range and d < bestDist then
					bestDist = d
					bestOrb = obj
				end
			end
		end
	end
	if bestOrb then
		pcall(function()
			my.CFrame = CFrame.new(bestOrb.Position + Vector3.new(0, 2.5, 0))
		end)
	end
end

local function collectOrbLuck()
	-- dedicated luck orb / event orb pickup
	fireNames({
		"CollectOrb", "CollectOrbs", "ClaimOrb", "PickupOrb", "Orbs_Collect",
		"CollectAllOrbs", "MagnetOrbs", "CollectLuckOrb", "LuckOrb",
		"ClaimLuck", "PickupLuck", "CollectLuckyOrb", "LuckyOrb",
		"OrbMagnet", "Magnet",
	}, {})
	local my = hrp()
	if not my then return end
	local roots = {
		Workspace:FindFirstChild("__THINGS"),
		Workspace:FindFirstChild("Orbs"),
		Workspace:FindFirstChild("Rendered"),
		Workspace,
	}
	local range = (S.collectRange or 120) * 1.5
	local targets = {}
	for _, root in ipairs(roots) do
		if root then
			for _, obj in ipairs(root:GetDescendants()) do
				if obj:IsA("BasePart") or obj:IsA("MeshPart") then
					local n = obj.Name:lower()
					if n:find("orb") or n:find("luck") or n:find("lucky") or n:find("burst") then
						local d = (obj.Position - my.Position).Magnitude
						if d < range then
							table.insert(targets, {part = obj, dist = d})
						end
					end
				end
			end
		end
	end
	table.sort(targets, function(a, b) return a.dist < b.dist end)
	for i = 1, math.min(5, #targets) do
		local t = targets[i]
		pcall(function()
			my.CFrame = CFrame.new(t.part.Position + Vector3.new(0, 2, 0))
		end)
		fireNames({"CollectOrb", "PickupOrb", "ClaimOrb", "Magnet"}, {})
		task.wait(0.05)
	end
end
