local function findGlobalFunction(...)
    for i = 1, select("#", ...) do
        local name = select(i, ...)
        local v = rawget(env, name)
        if type(v) == "function" then return v end
        v = rawget(_G, name)
        if type(v) == "function" then return v end
    end
    return nil
end

local function setHidden(instance, property, value)
    if not instance then return false end
    local setter = findGlobalFunction("sethiddenproperty", "set_hidden_property", "sethiddenprop")
    if setter then
        local ok = pcall(setter, instance, property, value)
        if ok then return true end
    end
    return pcall(function() instance[property] = value end)
end

local function getHidden(instance, property)
    if not instance then return false, nil end
    local getter = findGlobalFunction("gethiddenproperty", "get_hidden_property", "gethiddenprop")
    if getter then
        local ok, value = pcall(getter, instance, property)
        if ok then return true, value end
    end
    local ok, value = pcall(function() return instance[property] end)
    return ok, value
end

local function isBasePart(inst)
    if not inst then return false end
    local ok, r = pcall(function() return inst:IsA("BasePart") end)
    return ok and r == true
end

    if ok and isBasePart(root) then return root end
    return nil
end

local function destroyFakeRoot()
    local fake = AH.fakeRoot
    AH.fakeRoot = nil
    if fake then pcall(function() fake:Destroy() end) end
end

local function createFakeRoot(basePosition)
    destroyFakeRoot()
    local fake = Instance.new("Part")
    fake.Name = FAKE_ROOT_NAME
    fake.Size = Vector3.new(2, 2, 1)
    fake.Anchored = true
    fake.CanCollide = false
    fake.CanTouch = false
    fake.CanQuery = false
    fake.Transparency = 1
    fake.CFrame = CFrame.new(basePosition.X, FAKE_ROOT_Y, basePosition.Z)
    fake.AssemblyLinearVelocity = FAKE_ROOT_VELOCITY
    fake.Parent = Workspace
    AH.fakeRoot = fake
    return fake
end

local function fakeRootIsUsable()
    local fake = AH.fakeRoot
    if not isBasePart(fake) then return false end
    local ok, parent = pcall(function() return fake.Parent end)
    return ok and parent ~= nil
end

local function assignFakeReplicationRoot(rootPart, fake)
    if not isBasePart(rootPart) or not isBasePart(fake) then return false end
    AH.repRootOwner = rootPart
    return setHidden(rootPart, "PhysicsRepRootPart", fake)
end

local function restoreReplicationRoot()
    local root = AH.rootPart or getCurrentRoot()
    if isBasePart(root) then setHidden(root, "PhysicsRepRootPart", root) end
    if isBasePart(AH.repRootOwner) and AH.repRootOwner ~= root then
        setHidden(AH.repRootOwner, "PhysicsRepRootPart", AH.repRootOwner)
    end
    AH.repRootOwner = nil
end

    if not root then return end
    if not fakeRootIsUsable() then
        local fake = createFakeRoot(root.Position)

        return

    local fake = AH.fakeRoot
    pcall(function()
        fake.Anchored = true
        fake.AssemblyLinearVelocity = FAKE_ROOT_VELOCITY
    end)
    local got, current = getHidden(root, "PhysicsRepRootPart")
    if not got or current ~= fake then
        setHidden(root, "PhysicsRepRootPart", fake)
    end
end

local trackers = {}
local isAnchored = false
local lastAnchorParts = {}
local detectionConn

local function CheckTeleport(plr, root)
    local now = tick()
    local prev = trackers[plr]
    if not prev then
        trackers[plr] = { lastPos = root.Position, lastTime = now }
        return false
    end
    local dt = now - prev.lastTime
    if dt < 0.005 then return false end
    local delta = (root.Position - prev.lastPos).Magnitude
    local speed = delta / dt
    prev.lastPos = root.Position
    prev.lastTime = now
    return speed > AH.config.VelocityThreshold or delta > AH.config.DeltaThreshold
end

local function HasTool(plr)
    local c = plr and plr.Character
    return c and c:FindFirstChildOfClass("Tool") ~= nil
end

local function AnchorSelf()
    if not AH.config.AnchorEnabled or isAnchored then return end
    isAnchored = true
    local char = LP.Character
    if not char then isAnchored = false return end
    lastAnchorParts = {}
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            table.insert(lastAnchorParts, { part = part, anchored = part.Anchored })
            part.Anchored = true
        end
    end
    task.delay(AH.config.AnchorTime, function()
        for _, entry in ipairs(lastAnchorParts) do
            if entry.part and entry.part.Parent then
                entry.part.Anchored = entry.anchored
            end
        end
        lastAnchorParts = {}
        isAnchored = false
    end)
end

local function PushAttacker(attacker, myRoot)
    if not AH.config.PushEnabled then return end
    local atkRoot = GetRoot(attacker)
    if not atkRoot then return end
    local dir = atkRoot.Position - myRoot.Position
    if dir.Magnitude < 0.1 then
        dir = Vector3.new(math.random(-1, 1), 0, math.random(-1, 1))
    end
    dir = dir.Unit
    if dir.Y ~= dir.Y then dir = Vector3.new(1, 0, 0) end
    pcall(function() atkRoot:SetNetworkOwner(LP) end)
    local force = dir * AH.config.PushPower + Vector3.new(0, 100, 0)
    pcall(function() atkRoot.AssemblyLinearVelocity = force end)
    pcall(function()
        local old = atkRoot:FindFirstChild("EncryptedHub_Push")
        if old then old:Destroy() end
        local oldAtt = atkRoot:FindFirstChild("EncryptedHub_Att")
        if oldAtt then oldAtt:Destroy() end
        local att = Instance.new("Attachment")
        att.Name = "EncryptedHub_Att"
        att.Parent = atkRoot
        local lv = Instance.new("LinearVelocity")
        lv.Name = "EncryptedHub_Push"
        lv.Attachment0 = att
        lv.MaxForce = math.huge
        lv.VectorVelocity = force
        lv.RelativeTo = Enum.ActuatorRelativeTo.World
        lv.Parent = atkRoot
        task.delay(0.25, function()
            if lv and lv.Parent then lv:Destroy() end
            if att and att.Parent then att:Destroy() end
        end)
    end)
end

local function Activate(attacker)
    local now = tick()
    if now - AH.config.LastTrigger < AH.config.Cooldown then return end
    AH.config.LastTrigger = now
    local myRoot = GetRoot(LP)
    if not myRoot then return end
    Flash()
    if attacker then PushAttacker(attacker, myRoot) end
    AnchorSelf()
end

local function StartDetection()
    if detectionConn then detectionConn:Disconnect() end
    detectionConn = RunService.Heartbeat:Connect(function()
        if not AH.enabled or isAnchored then return end
        local myRoot = GetRoot(LP)
        if not myRoot then return end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP then
                local root = GetRoot(plr)
                if root then
                    local dist = (root.Position - myRoot.Position).Magnitude
                    if dist <= AH.config.Radius then
                        if CheckTeleport(plr, root) then
                            Activate(plr)
                            return
                        end
                        if dist < 6 and HasTool(plr) then
                            Activate(plr)
                            return
                        end
                    else
                        CheckTeleport(plr, root)
                    end
                end
            end
        end
    end)
end

local function StopDetection()
    if detectionConn then detectionConn:Disconnect() detectionConn = nil end
end

local function bindCharacter(character)
    local oldRoot = AH.rootPart
    AH.character = character
    AH.rootPart = getCurrentRoot(character)
    if AH.enabled then
        if isBasePart(oldRoot) and oldRoot ~= AH.rootPart then
            setHidden(oldRoot, "PhysicsRepRootPart", oldRoot)
        end
        destroyFakeRoot()
        local root = AH.rootPart
        if not root and character then
            local ok, waited = pcall(function()
                return character:WaitForChild("HumanoidRootPart", 8)
            end)
            if ok and isBasePart(waited) then
                root = waited
                AH.rootPart = root
            end
        end
        if root then
            AH.originalCFrame = root.CFrame
            local fake = createFakeRoot(root.Position)
            assignFakeReplicationRoot(root, fake)
            startStep()
            applyModeToFake()
        end
    end
end

local function setAntiHit(value)
    if not AH.alive then return false end
    value = value == true
    if AH.enabled == value then return value end
    if value then
        local root = getCurrentRoot(AH.character or LP.Character)
        if not root then
            notify("Anti-Hit", "No character / HRP")
            return false
        end
        AH.rootPart = root
        AH.character = LP.Character
        AH.originalCFrame = root.CFrame
        AH.enabled = true
        local fake = createFakeRoot(root.Position)
        assignFakeReplicationRoot(root, fake)
        startStep()
        StartDetection()
        if AH.mode == "ORIGINAL" or AH.config.FlashEnabled then
            startLoop()
        else
            applyModeToFake()
        end
        notify("Anti-Hit", "ON · " .. AH.mode)
    else
        AH.enabled = false
        stopLoop()
        StopDetection()
        stopStep()
        restoreReplicationRoot()
        destroyFakeRoot()
        local root = AH.rootPart or getCurrentRoot(AH.character)
        if isBasePart(root) then
            pcall(function() root:SetNetworkOwner(LP) end)
        end
        AH.originalCFrame = nil
        notify("Anti-Hit", "OFF")
    end
    return AH.enabled
end

local function setMode(name)
    AH.mode = name
    if AH.enabled then
        stopLoop()
        if AH.mode == "ORIGINAL" or AH.config.FlashEnabled then
            startLoop()
        else
            applyModeToFake()
        end
    end
end

        clearESP()

        for _, plr in ipairs(Players:GetPlayers()) do

                local char = plr.Character
                if char then

                        if role == "Sheriff" and not F.espSheriff then show = false end

                        bb.Adornee = hrp

local function getRole(plr)
	-- MM2 common patterns
	local role = nil
	pcall(function()
		local ls = plr:FindFirstChild("leaderstats") or plr:FindFirstChild("PlayerData")
		if plr:GetAttribute("Role") then role = plr:GetAttribute("Role") end
		if plr:GetAttribute("role") then role = plr:GetAttribute("role") end
	end)
	-- backpack / character tools
	local char = plr.Character
	local pack = plr:FindFirstChildOfClass("Backpack")
	local function has(name)
		name = name:lower()
		if char then
			for _, t in ipairs(char:GetChildren()) do
				if t:IsA("Tool") and t.Name:lower():find(name) then return true end
			end
		end
		if pack then
			for _, t in ipairs(pack:GetChildren()) do
				if t:IsA("Tool") and t.Name:lower():find(name) then return true end
			end
		end
		return false
	end
	if has("knife") then return "Murderer" end
	if has("gun") then return "Sheriff" end
	-- ReplicatedStorage role maps
	pcall(function()
		local rs = game:GetService("ReplicatedStorage")
		for _, folder in ipairs({rs:FindFirstChild("Role"), rs:FindFirstChild("Roles"), rs:FindFirstChild("GameData")}) do
			if folder then
				local v = folder:FindFirstChild(plr.Name) or folder:FindFirstChild(tostring(plr.UserId))
				if v then
					if v:IsA("StringValue") or v:IsA("StringValue") then role = v.Value end
					if v:IsA("StringValue") then role = v.Value end
					if typeof(v) == "Instance" and v:GetAttribute("Role") then role = v:GetAttribute("Role") end
				end
			end
		end
	end)
	return role or "Innocent"
end

local function isMurderer(plr) return tostring(getRole(plr)):lower():find("murder") ~= nil end
local function isSheriff(plr) return tostring(getRole(plr)):lower():find("sheriff") ~= nil or tostring(getRole(plr)):lower():find("hero") ~= nil end

local function equipTool(substr)
	local char = LP.Character
	local pack = LP:FindFirstChildOfClass("Backpack")
	substr = substr:lower()
	local function try(container)
		if not container then return false end
		for _, t in ipairs(container:GetChildren()) do
			if t:IsA("Tool") and t.Name:lower():find(substr) then
				pcall(function()
					local hum = getHum()
					if hum then hum:EquipTool(t) end
				end)
				return true
			end
		end
		return false
	end
	return try(char) or try(pack)
end

local function swingKnife()
	equipTool("knife")
	local char = LP.Character
	if not char then return end
	local tool = char:FindFirstChildOfClass("Tool")
	if tool then
		pcall(function() tool:Activate() end)
		pcall(function()
			local remote = tool:FindFirstChildWhichIsA("RemoteEvent", true)
			if remote then remote:FireServer("Slash") end
		end)
	end
	-- common MM2 knife remotes
	pcall(function()
		local rs = game:GetService("ReplicatedStorage")
		for _, name in ipairs({"ShootGun", "KnifeKill", "KillPlayer", "Slash"}) do
			local r = rs:FindFirstChild(name, true)
			if r and r:IsA("RemoteEvent") then r:FireServer() end
		end
	end)
end

		if r and targetRoot then

		end
	end)
end

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((val - min) / (max - min), 0, 1, 0)
    fill.BackgroundColor3 = COLORS.accent
    fill.BorderSizePixel = 0
    fill.Parent = bar
    corner(fill, 4)
    local dragging = false
    local function update(x)
        local rel = math.clamp((x - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
        val = math.floor(min + (max - min) * rel + 0.5)
        fill.Size = UDim2.new(rel, 0, 1, 0)
        t.Text = label .. ": " .. tostring(val)
        if callback then callback(val) end
    end
    bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            update(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            update(input.Position.X)
        end
    end)
end

local function mm2Equip(sub)
	sub = sub:lower()
	local char, pack = LP.Character, LP:FindFirstChildOfClass("Backpack")
	for _, c in ipairs({char, pack}) do
		if c then
			for _, t in ipairs(c:GetChildren()) do
				if t:IsA("Tool") and t.Name:lower():find(sub) then
					pcall(function() local h=getHum() if h then h:EquipTool(t) end end)
					return t
				end
			end
		end
	end
end

local function mm2FireTool()
	local char = LP.Character
	local tool = char and char:FindFirstChildOfClass("Tool")
	if tool then pcall(function() tool:Activate() end) end
end

local function nearestPlayer(pred, maxDist)
	local my = getHRP()
	if not my then return nil end
	local best, bd = nil, maxDist or 1e9
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LP and p.Character then
			local okPred = true
			if pred then
				local ok, res = pcall(pred, p)
				okPred = ok and res
			end
			if okPred then
				local r = p.Character:FindFirstChild("HumanoidRootPart")
				local h = p.Character:FindFirstChildOfClass("Humanoid")
				if r and h and h.Health > 0 then
					local d = (r.Position - my.Position).Magnitude
					if d < bd then best, bd = p, d end
				end
			end
		end
	end
	return best, bd
end

			local t, d = nearestPlayer(function(p) return isMurderer and isMurderer(p) end, F1K.i_alertDist)
			if t and d and d < F1K.i_alertDist then
				if not F1K._alerted then
					F1K._alerted = true
					notify("DANGER", "Murderer nearby: " .. t.Name)
				end
			else
				F1K._alerted = false
			end
		end

		-- movement presets
		if hum then
			for _, pair in ipairs({
				{"spd_16", 16}, {"spd_18", 18}, {"spd_20", 20}, {"spd_22", 22}, {"spd_24", 24},
				{"spd_26", 26}, {"spd_28", 28}, {"spd_30", 30}, {"spd_32", 32}, {"spd_35", 35},
				{"spd_40", 40}, {"spd_45", 45}, {"spd_50", 50}, {"spd_55", 55}, {"spd_60", 60},
			}) do
				if f1k(pair[1]) then hum.WalkSpeed = pair[2] end
			end
			for _, pair in ipairs({
				{"jp_50", 50}, {"jp_60", 60}, {"jp_70", 70}, {"jp_80", 80}, {"jp_90", 90},
				{"jp_100", 100}, {"jp_110", 110}, {"jp_120", 120}, {"jp_140", 140}, {"jp_160", 160},
			}) do
				if f1k(pair[1]) then hum.JumpPower = pair[2] end
			end
		end

		-- coin helpers
		if f1k("coin_magnet") or f1k("coin_pull_all") then
			for _, d in ipairs(Workspace:GetDescendants()) do
				if d:IsA("BasePart") and d.Name:lower():find("coin") then
					pcall(function() d.CFrame = my.CFrame end)
				end
			end
		end
		if f1k("coin_tp_one") then
			for _, d in ipairs(Workspace:GetDescendants()) do
				if d:IsA("BasePart") and d.Name:lower():find("coin") then
					pcall(function() my.CFrame = d.CFrame + Vector3.new(0,2,0) end)
					break
				end
			end
		end

		-- protect
		if f1k("prot_antistun") and hum then
			hum.PlatformStand = false
			hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
			hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
		end
		if f1k("prot_heal") and hum then hum.Health = hum.MaxHealth end
		if f1k("prot_nofall") and hum then
			hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
		end

		-- knife reach / hitbox
		if F1K.knifeReach or f1k("knife_reach") then
			local char = LP.Character
			local tool = char and char:FindFirstChildOfClass("Tool")
			if tool and tool.Name:lower():find("knife") then
				for _, p in ipairs(tool:GetDescendants()) do
					if p:IsA("BasePart") then
						p.Size = Vector3.new(F1K.knifeSize, F1K.knifeSize, F1K.knifeSize)
						p.Massless = true
						p.Transparency = 0.75
					end
				end
			end
		end
		if F1K.hitbox or f1k("hitbox_expand") then
			for _, p in ipairs(Players:GetPlayers()) do
				if p ~= LP and p.Character then
					local hrp = p.Character:FindFirstChild("HumanoidRootPart")
					if hrp then
						pcall(function()
							hrp.Size = Vector3.new(F1K.hitboxSize, F1K.hitboxSize, F1K.hitboxSize)
							hrp.Transparency = 0.7
							hrp.CanCollide = false
						end)
					end
				end
			end
		end
		end -- if my
	end -- while
end)
