-- ===================== REMOTE ENGINE =====================
local function collectRemotes()
	local list = {}
	local function add(inst)
		if not inst then return end
		if inst:IsA("RemoteEvent") or inst:IsA("RemoteFunction") then
			table.insert(list, inst)
		end
	end
	for _, d in ipairs(ReplicatedStorage:GetDescendants()) do
		if d:IsA("RemoteEvent") or d:IsA("RemoteFunction") then
			add(d)
		end
	end
	-- common numbered remotes used by CE scripts
	for i = 1, 12 do
		add(ReplicatedStorage:FindFirstChild("RemoteEvent_" .. i))
		add(ReplicatedStorage:FindFirstChild("RemoteFunction_" .. i))
		add(ReplicatedStorage:FindFirstChild("RE_" .. i))
		add(ReplicatedStorage:FindFirstChild("RF_" .. i))
	end
	S.remotes = list
	return list
end

local function fireAll(actionNames, args)
	args = args or {}
	local remotes = S.remotes
	if not remotes or #remotes == 0 then
		remotes = collectRemotes()
	end
	for _, name in ipairs(actionNames) do
		for _, remote in ipairs(remotes) do
			pcall(function()
				if remote:IsA("RemoteEvent") then
					remote:FireServer(name, unpack(args))
					remote:FireServer(name, args)
					if #args > 0 then
						remote:FireServer(unpack(args))
					end
				elseif remote:IsA("RemoteFunction") then
					remote:InvokeServer(name, unpack(args))
					remote:InvokeServer(name, args)
				end
			end)
		end
		-- also try Packages/Networking style modules
		pcall(function()
			local net = ReplicatedStorage:FindFirstChild("Networking", true)
				or ReplicatedStorage:FindFirstChild("Network", true)
				or ReplicatedStorage:FindFirstChild("Remotes", true)
			if net then
				local r = net:FindFirstChild(name, true)
				if r then
					if r:IsA("RemoteEvent") then r:FireServer(unpack(args)) end
					if r:IsA("RemoteFunction") then r:InvokeServer(unpack(args)) end
				end
			end
		end)
	end
end

local function getMyCountry()
	-- try player attributes / values
	local c = nil
	pcall(function()
		c = LP:GetAttribute("Country") or LP:GetAttribute("country")
	end)
	if c and tostring(c) ~= "" then
		S.myCountry = tostring(c)
		return S.myCountry
	end
	pcall(function()
		local v = LP:FindFirstChild("Country") or LP:FindFirstChild("country")
		if v and v:IsA("ValueBase") then
			S.myCountry = tostring(v.Value)
		end
	end)
	if S.myCountry then return S.myCountry end
	-- scan regions for leaderstats-like ownership
	pcall(function()
		local folder = Workspace:FindFirstChild("Regions")
			or Workspace:FindFirstChild("Tiles")
			or Workspace:FindFirstChild("Cities")
			or Workspace:FindFirstChild("Map")
		if not folder then return end
		-- can't know country without data - leave nil
	end)
	return S.myCountry
end

local function getOwnedRegions()
	local owned = {}
	local country = getMyCountry()
	local folders = {
		Workspace:FindFirstChild("Regions"),
		Workspace:FindFirstChild("Tiles"),
		Workspace:FindFirstChild("Cities"),
		Workspace:FindFirstChild("Map"),
		Workspace:FindFirstChild("Countries"),
	}
	for _, folder in ipairs(folders) do
		if folder then
			for _, region in ipairs(folder:GetChildren()) do
				local ok = false
				local c = region:FindFirstChild("Country") or region:FindFirstChild("Owner") or region:FindFirstChild("owner")
				if c and country and tostring(c.Value) == tostring(country) then
					ok = true
				end
				if region:GetAttribute("Country") and country and tostring(region:GetAttribute("Country")) == tostring(country) then
					ok = true
				end
				if ok then
					table.insert(owned, region)
				end
			end
		end
	end
	return owned
end

local function getMySoldiers()
	local list = {}
	local country = getMyCountry()
	local folders = {
		Workspace:FindFirstChild("SoldiersFolder"),
		Workspace:FindFirstChild("Soldiers"),
		Workspace:FindFirstChild("Armies"),
		Workspace:FindFirstChild("Troops"),
		Workspace:FindFirstChild("Units"),
	}
	for _, folder in ipairs(folders) do
		if folder then
			for _, unit in ipairs(folder:GetChildren()) do
				local c = unit:FindFirstChild("Country") or unit:FindFirstChild("Owner")
				if c and country and tostring(c.Value) == tostring(country) then
					table.insert(list, unit)
				elseif unit:GetAttribute("Country") and country and tostring(unit:GetAttribute("Country")) == tostring(country) then
					table.insert(list, unit)
				end
			end
		end
	end
	return list
end

-- ===================== FEATURE ACTIONS =====================
local function spawnOnAllRegions()
	local owned = getOwnedRegions()
	local amount = S.spawnAmount or 50000
	local unitType = S.spawnType or "Soldier"
	local names = {
		"CreateArmyOnTile", "PlaceTroops", "Recruit", "SpawnSoldiers", "AddArmy",
		"SpawnArmy", "CreateArmy", "MultiSpawn", "SpawnUnit", "DeployArmy",
	}
	local n = 0
	if #owned == 0 then
		-- still fire generic spawn
		fireAll(names, {unitType, amount})
		fireAll(names, {amount, unitType})
		notify("Spawn", "No owned tiles found — fired generic")
		return
	end
	for _, region in ipairs(owned) do
		fireAll(names, {region, unitType, amount})
		fireAll(names, {region.Name, unitType, amount})
		fireAll(names, {region, amount})
		n = n + 1
		task.wait(S.spawnDelay or 0.35)
	end
	notify("Spawn", "Sent to " .. n .. " regions")
end

local function reinforceAll()
	local units = getMySoldiers()
	local names = {"AddToArmy", "AddUnits", "Reinforce", "SpawnSoldiers", "CreateArmyOnTile"}
	local n = 0
	for _, unit in ipairs(units) do
		local cur = unit:FindFirstChild("TotalAmount") or unit:FindFirstChild("Amount") or unit:FindFirstChild("Size")
		local amt = (cur and tonumber(cur.Value)) or 1000
		fireAll(names, {unit, S.spawnType or "Soldier", amt + (S.spawnAmount or 25000)})
		n = n + 1
		task.wait(0.2)
	end
	notify("Reinforce", n .. " stacks")
end

local function upgradeAllCities()
	local owned = getOwnedRegions()
	local names = {
		"UpgradeCity", "UpgradeTier", "UpgradeDefense", "Upgrade",
		"CityUpgrade", "DevelopCity", "MaxUpgrade", "UpgradeBuilding",
	}
	local n = 0
	if #owned == 0 then
		fireAll(names, {})
		fireAll({"UpgradeAll", "MassUpgrade", "UpgradeEverything"}, {})
		notify("Upgrade", "Generic upgrade fired")
		return
	end
	for _, region in ipairs(owned) do
		fireAll(names, {region})
		fireAll(names, {region.Name})
		fireAll(names, {region, "Tier"})
		fireAll(names, {region, "Defense"})
		n = n + 1
		task.wait(S.upgradeDelay or 0.5)
	end
	notify("Upgrade", n .. " cities")
end

			dragging = true
			dragStart = input.Position
			startPos = Main.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then dragging = false end
			end)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local d = input.Position - dragStart
			Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end)

		if gp then return end
		if input.KeyCode == Enum.KeyCode.RightControl and unlocked then
			Main.Visible = not Main.Visible
		end
	end)

end
