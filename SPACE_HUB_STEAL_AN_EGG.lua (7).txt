-- Eggs
local function collectEggs()
	local list={}
	if EggState and type(EggState.ReadFieldEggs)=="function" then
		local s,data=pcall(function() return EggState.ReadFieldEggs() end)
		if s and type(data)=="table" then
			for uid,d in pairs(data) do
				if type(d)=="table" then
					local model=d.model or d.Model
					local rarity=d.rarity or d.Rarity or (d.record and (d.record.Rarity or d.record.Category)) or "Common"
					local part=model and model:FindFirstChildWhichIsA("BasePart",true)
					local val=tonumber(d.value or d.Value or d.price or 0) or 0
					if part then
						table.insert(list,{uid=uid,model=model,part=part,rarity=tostring(rarity),value=val,score=rarityScore(rarity)*1e6+val+part.Size.Magnitude})
					end
				end
			end
		end
	end
	for _,d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("ProximityPrompt") and d.Enabled then
			local n=d.Name
			if n=="CarryAreaEgg" or n:lower():find("carry") or n:lower():find("egg") then
				local model=d:FindFirstAncestorOfClass("Model")
				local part=d.Parent
				if part and part:IsA("BasePart") then
					local already=false
					for _,e in ipairs(list) do if e.part==part then already=true break end end
					if not already then
						local rname=(model and (model:GetAttribute("Rarity") or model.Name)) or part.Name
						table.insert(list,{model=model,part=part,prompt=d,rarity=tostring(rname),value=0,score=rarityScore(rname)*1e6+part.Size.Magnitude})
					end
				end
			end
		end
	end
	if F.priority=="Highest" then
		table.sort(list,function(a,b) return (a.score or 0)>(b.score or 0) end)
	elseif F.priority=="Nearest" then
		local hrp=getHRP()
		if hrp then
			table.sort(list,function(a,b)
				local da=a.part and (a.part.Position-hrp.Position).Magnitude or 1e9
				local db=b.part and (b.part.Position-hrp.Position).Magnitude or 1e9
				return da<db
			end)
		end
	end
	return list
end

local function guardDist(pos)
	local best = math.huge
	for _, g in ipairs(findGuards()) do

-- Chilli r1594 SeparationLine targeting
local function agLinePos(cfg, fromPos)
	local world = Workspace:FindFirstChild("World") or Workspace:FindFirstChild("__OBJECTS")
	local areas = world and world:FindFirstChild("Areas")
	local sep = areas and areas:FindFirstChild("SeparationLine")
	if sep and sep:IsA("BasePart") then
		local cf = sep.CFrame
		local axis = (Vector3.new(0, 1, 0)):Cross(sep.Size.X >= sep.Size.Z and cf.RightVector or cf.LookVector)
		local flat = Vector3.new(axis.X, 0, axis.Z)
		if flat.Magnitude > 0.001 then
			local unit = flat.Unit
			local side = ((fromPos - cf.Position):Dot(unit) >= 0) and -unit or unit
			local p = cf.Position + side * (tonumber(cfg.LineOffset) or 8)
			return Vector3.new(p.X, fromPos.Y + 0.5, p.Z)
		end
	end
	return nil
end

local function agTargetPos(cfg, startPos)
	local t = tostring(cfg.Target or "line")
	if t == "sky" then return startPos + Vector3.new(0, tonumber(cfg.Height) or 45, 0) end
	if t == "point" and typeof(cfg.Point) == "Vector3" then return cfg.Point end
	if t == "line" then
		local p = agLinePos(cfg, startPos)
		if p then return p end
	end
	return agHomePos()
end

local function agOffsetPos(cfg, startPos)
	local base = agTargetPos(cfg, startPos)
	return base + Vector3.new(tonumber(cfg.OffsetX) or 0, tonumber(cfg.Height) or 0, tonumber(cfg.OffsetZ) or 0)
end

end

-- Chilli r1600 sequence: limp + step path home/start then release
local function agRunSequence(startedAt)
	local cfg = agGetCfg()
	local hrp = getHRP()
	local hum = getHum()
	if not hrp or not hum then
		AG.Busy = false
		AG.Active = false
		return
	end
	local startPos = hrp.Position
	local homeish = agOffsetPos(cfg, startPos)
	local wasStand = hum.PlatformStand
	F.status = "Anti Guard is slipping past the guard"
	AG.Busy = true
	AG.BusySince = os.clock()
	AG.Active = true

	-- Limp (Chilli)
	if cfg.Limp ~= false then
		pcall(function() hum.PlatformStand = true end)
	end

	local steps = type(cfg.Steps) == "table" and cfg.Steps or {}
	local releaseAt = tonumber(cfg.ReleaseAt) or 0.8
	local freeze = cfg.Freeze == true

	local function waitUntil(at)
		while agStillActive() and (os.clock() - startedAt) < at do
			RunService.Heartbeat:Wait()
			if freeze then
				pcall(function()
					local h = getHRP()
					if h then h.AssemblyLinearVelocity = Vector3.zero h.AssemblyAngularVelocity = Vector3.zero end
				end)
			end
		end
		return agStillActive()
	end

	local last = startPos
	for _, step in ipairs(steps) do
		if type(step) ~= "table" then break end
		local at = tonumber(step.At) or 0
		if not waitUntil(at) then break end
		local dest = (step.To == "start") and startPos or homeish
		-- jitter
		local j = tonumber(cfg.Jitter) or 0
		if j > 0 then
			dest = dest + Vector3.new((math.random()*2-1)*j, 0, (math.random()*2-1)*j)
		end
		local face = CFrame.new()
		if tostring(cfg.Facing) == "Spin" then
			face = CFrame.Angles(0, math.rad(math.random(0, 359)), 0)
		end
		agSetRoot(CFrame.new(dest) * face, freeze)
		last = dest
		F.status = "Anti Guard step " .. tostring(step.To or "?")
	end

	waitUntil(releaseAt)

	-- rise / finish
	F.status = "Anti Guard done, rising up"
	pcall(function()
		local h = getHum()
		if h then h.PlatformStand = wasStand end
	end)
	-- soft up
	local hrp2 = getHRP()
	if hrp2 then
		pcall(function()
			hrp2.CFrame = hrp2.CFrame + Vector3.new(0, 4, 0)
		end)
	end

	AG.Busy = false
	AG.Active = false
	F.status = isCarrying() and "Anti Guard done, getting ready" or "Anti Guard done"
end

local function agStartIfNeeded()
	if not AG.Enabled or not F.antiGuard then return end
	if AG.Active or AG.Busy then return end
	if not AG.Carrying then return end
	-- HitArms lockout (Chilli r1603)
	if (tonumber(AG.HitArms) or 0) > 0 then
		if os.clock() - (tonumber(AG.HitArmedAt) or 0) <= 25 then
			return
		end
		AG.HitArms = 0
	end
	F.status = "Waiting for Anti Guard to start"
	AG.Busy = true
	AG.BusySince = os.clock()
	task.spawn(function()
		local ok, err = pcall(agRunSequence, os.clock())
		if not ok then
			warn("[Space Hub] AG error", err)
			AG.Busy = false
			AG.Active = false
			local h = getHum()
			if h then pcall(function() h.PlatformStand = false end) end
		end
	end)
end

local weldAcc = 0

	if not F.antiGuard or not AG.Enabled then return end
	-- busy timeout
	if AG.Busy then
		local lim = math.max(tonumber(agGetCfg().BusyLimit) or 2.5, (tonumber(agGetCfg().ReleaseAt) or 0) + 1)
		if os.clock() - (AG.BusySince or 0) > lim then
			AG.Busy = false

			local h = getHum()

		end

	if weldAcc < (tonumber(agGetCfg().WeldScanGap) or 0.03) then return end
	weldAcc = 0
	local w = agScanWeldCarry()
	if w ~= AG.WeldCarrying then
		AG.WeldCarrying = w
		AG.Carrying = AG.SignalCarrying or AG.WeldCarrying
		agStartIfNeeded()
	end
end))

local function antiGuardStep()
	-- manual flee remains available
	if not F.antiGuard then return end
	local hrp = getHRP()
	if not hrp then return end
	if guardDist(hrp.Position) < 14 and not AG.Busy then
		local nearest, nd = nil, math.huge
		for _, g in ipairs(findGuards()) do
			local d = (g.Position - hrp.Position).Magnitude
			if d < nd then nd, nearest = d, g end
		end
		if nearest then
			local away = (hrp.Position - nearest.Position)
			if away.Magnitude > 0.1 then
				away = away.Unit * 12
				pcall(function() hrp.CFrame = CFrame.new(hrp.Position + Vector3.new(away.X, 3, away.Z)) end)
			end
		end
	end
end

	if F2.pathMethod=="Walk" then
		local target=entry.part.Position+Vector3.new(0,3,0)
		for i=1,6 do local h=getHRP() if not h then break end h.CFrame=h.CFrame:Lerp(CFrame.new(target),0.35) task.wait(0.05) end
		pcall(function() local h=getHRP() if h then h.CFrame=CFrame.new(target) end end)
	else pcall(function() hrp.CFrame=entry.part.CFrame+Vector3.new(0,3,0) end) end
	task.wait(0.1)
	local prompt=entry.prompt
	if not prompt and entry.model then
		prompt=entry.model:FindFirstChild("CarryAreaEgg",true)
		if not prompt then for _,d in ipairs(entry.model:GetDescendants()) do if d:IsA("ProximityPrompt") then prompt=d break end end end
	end
	if prompt then firePrompt(prompt) task.wait(F.stealDelay) return true end
	for _,d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("ProximityPrompt") and d.Enabled then
			local p=d.Parent
			if p and p:IsA("BasePart") and (p.Position-hrp.Position).Magnitude<14 then
				firePrompt(d) task.wait(F.stealDelay) return true
			end
		end
	end
	return false
end

local function safeDeliver()
	if not F.safeDelivery then local b=findBase() if b then goTo(b) end return end
	F.status="Safe delivery"
	for _=1,10 do
		local hrp=getHRP() if not hrp then break end
		if guardDist(hrp.Position)>22 then break end
		antiGuardStep() task.wait(0.35)
	end
	local base=findBase() if base then goTo(base) end
	task.wait(0.35)
end

local function stealBest()
	F.status="Scanning"
	local eggs=collectEggs()
	local need=RarityRank[F.minRarity] or 3
	for _,e in ipairs(eggs) do
		if rarityScore(e.rarity)>=need then
			F.status="Steal "..tostring(e.rarity)
			if stealEgg(e) then
				task.wait(0.3)
				if isCarrying() then
					if F.carrySpeed>16 then local h=getHum() if h then h.WalkSpeed=F.carrySpeed end end
					safeDeliver()
					return true
				end
			end
		end
	end
	if #eggs>0 then
		stealEgg(eggs[1]) task.wait(0.35)
		if isCarrying() then safeDeliver() return true end
	end
	return false
end

local function wearTM() invoke("RF/Treadmill/AskWearStill") end
local function doffTM() invoke("RF/Treadmill/AskDoff") end
local function farmTM(sec)
	local tm=findTreadmill() if tm then goTo(tm) end
	wearTM()
	local t0=os.clock()
	while F.alive and (F.autoFarm or F.autoTreadmill) and os.clock()-t0<sec do
		local hum=getHum()
		if hum then pcall(function() hum.WalkSpeed=math.max(hum.WalkSpeed,F.speedVal) hum:Move(Vector3.new(1,0,0),true) end) end
		local hrp=getHRP()
		if hrp and tm and (hrp.Position-tm.Position).Magnitude>30 then goTo(tm) if F.stayTreadmill then wearTM() end end
		task.wait(0.25)
	end
end
local function upgradeTM()
	invoke("RF/Treadmill/AskTierRaise")
	-- try common variants
	invoke("RF/Treadmill/AskUpgrade")
end

local function runAutoFarm()
	while F.alive and F.autoFarm do
		F.status="Steal phase" stealBest()
		if not F.autoFarm then break end
		if F.autoUpgradeTM then upgradeTM() end
		F.status="Treadmill" farmTM(26)
		if not F.autoFarm then break end
		F.status="Wait restock" task.wait(1.2)
	end
	F.status="Idle"
end

local function slider(page,label,order,min,max,default,cb)
	local row=Instance.new("Frame")
	row.Size=UDim2.new(1,-8,0,42) row.BackgroundColor3=Color3.fromRGB(18,20,38) row.BorderSizePixel=0 row.LayoutOrder=order row.Parent=page corner(row,6)
	local val=default
	local t=Instance.new("TextLabel")
	t.Size=UDim2.new(1,-12,0,16) t.Position=UDim2.fromOffset(8,3) t.BackgroundTransparency=1
	t.Text=label..": "..tostring(val) t.TextColor3=COLORS.text t.Font=Enum.Font.Gotham t.TextSize=11 t.TextXAlignment=Enum.TextXAlignment.Left t.Parent=row
	local bar=Instance.new("Frame") bar.Size=UDim2.new(1,-16,0,7) bar.Position=UDim2.fromOffset(8,26) bar.BackgroundColor3=Color3.fromRGB(30,35,60) bar.BorderSizePixel=0 bar.Parent=row corner(bar,4)
	local fill=Instance.new("Frame") fill.Size=UDim2.new((val-min)/math.max(max-min,1),0,1,0) fill.BackgroundColor3=COLORS.accent fill.BorderSizePixel=0 fill.Parent=bar corner(fill,4)
	local dragging=false
	local function update(x)
		local rel=math.clamp((x-bar.AbsolutePosition.X)/math.max(bar.AbsoluteSize.X,1),0,1)
		val=math.floor(min+(max-min)*rel+0.5) fill.Size=UDim2.new(rel,0,1,0) t.Text=label..": "..tostring(val)
		if cb then pcall(cb,val) end
	end
	bar.InputBegan:Connect(function(input)
		if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then dragging=true update(input.Position.X) end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then dragging=false end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType==Enum.UserInputType.MouseMovement or input.UserInputType==Enum.UserInputType.Touch) then update(input.Position.X) end
	end)
end
