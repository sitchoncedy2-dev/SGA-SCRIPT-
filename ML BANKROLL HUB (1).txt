local function formatNumber(number)\r\
    if not number then return \"0\" end\r\
    if number >= 1e15 then return string.format(\"%.2fQ\", number/1e15)\r\
    elseif number >= 1e12 then return string.format(\"%.2fT\", number/1e12)\r\
    elseif number >= 1e9 then return string.format(\"%.2fB\", number/1e9)\r\
    elseif number >= 1e6 then return string.format(\"%.2fM\", number/1e6)\r\
    elseif number >= 1e3 then return string.format(\"%.2fK\", number/1e3)\r\
    end\r\
    return tostring(math.floor(number))\r\
end\r\
\r\
local function formatTime(seconds)\r\
    local hours = math.floor(seconds / 3600)\r\
    local minutes = math.floor((seconds % 3600) / 60)\r\
    local secs = seconds % 60\r\
    return string.format(\"%02d:%02d:%02d\", hours, minutes, secs)\r\
end\r\
